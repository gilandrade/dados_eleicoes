1. Crosswalk cd_mun_tse ↔︎ cd_mun

O TSE usa um código de município próprio (CD_MUNICIPIO), diferente do código IBGE de 7 dígitos usado no resto da GeoReDUS. Não existe crosswalk em nenhum dos nossos repos (ver PLAN.md > Fontes de dados) — usamos a tabela pronta do projeto yuripassuelo/codigos_mun_tse_ibge, que já cruza os 5.570 municípios (sem o exterior).

caminho_crosswalk_bruto <- "dados/01_bruto/aux/relacao_mun_tse_ibge.csv"
dir.create(dirname(caminho_crosswalk_bruto), showWarnings = FALSE, recursive = TRUE)

if (!file.exists(caminho_crosswalk_bruto)) {
  req <- request(
    "https://raw.githubusercontent.com/yuripassuelo/codigos_mun_tse_ibge/master/relacao_mun_tse_ibge.csv"
  ) |>
    req_retry(max_tries = 3, backoff = \(n) 2^n) |>
    req_user_agent("dados_eleicoes (CEM/USP)")
  resp <- req_perform(req)
  writeBin(resp_body_raw(resp), caminho_crosswalk_bruto)
}

crosswalk_bruto <- read_csv(
  caminho_crosswalk_bruto,
  locale = locale(encoding = "latin1"),
  col_types = cols(.default = col_character())
)

crosswalk <- crosswalk_bruto |>
  transmute(
    # o CD_MUNICIPIO dos arquivos do TSE vem com 5 dígitos e zero à esquerda
    # (ex.: "01120"), mas o CSV de origem do crosswalk guarda cd_mun_tse sem
    # padding ("1120") — confirmado contra o arquivo real de locais em
    # 28/09/2026 (ver CLAUDE.md).
    cd_mun_tse = str_pad(cd_mun_tse, 5, pad = "0"),
    cd_mun = str_pad(cd_mun_ibge, 7, pad = "0"),
    sg_uf = SG_UF,
    nm_mun_ibge = nm_mun_ibge,
    nm_mun_tse = nm_mun_tse
  )

stopifnot(
  "crosswalk deveria ter 5.570 municípios (sem o exterior)" = nrow(crosswalk) == 5570,
  "cd_mun_tse não deveria ter duplicatas" = !anyDuplicated(crosswalk$cd_mun_tse),
  "cd_mun (IBGE) não deveria ter duplicatas" = !anyDuplicated(crosswalk$cd_mun)
)

write_parquet(crosswalk, file.path(AUX_DIR, "crosswalk_mun_tse_ibge.parquet"))

Checagem por nome (municípios novos/renomeados)

Nomes muito diferentes entre nm_mun_ibge e nm_mun_tse no mesmo cd_mun costumam indicar um município criado ou renomeado depois da tabela do crosswalk ter sido montada — checar à mão os que aparecerem aqui.

normaliza_texto <- function(x) {
  x |>
    stringr::str_to_upper() |>
    stringi::stri_trans_general("Latin-ASCII") |>
    stringr::str_squish()
}

divergencias <- crosswalk |>
  mutate(
    dist_nome = stringdist::stringdist(
      normaliza_texto(nm_mun_ibge), normaliza_texto(nm_mun_tse), method = "jw"
    )
  ) |>
  filter(dist_nome > 0.15) |>
  arrange(desc(dist_nome))

3. População adulta estimada por setor (pop18)

pop18(s) = total(s) − [0,4] − [5,9] − [10,14] − 0,6 × [15,19], a mesma conta de src/eleicoes/alocacao.pop18_por_setor() (Python) — ver PLAN.md > Método, passo 4. É calculada aqui (R, etapa “auxiliares”) e só lida em Python, para não duplicar a leitura dos CSVs do Censo em duas linguagens.

Os CSVs do cn22-pop já vêm com os indicadores nomeados no padrão GeoReDUS (cn22_pop01_res_tot_0, cn22_pop03_0a4_tot_0, …), não com as variáveis brutas do IBGE (V0001, V01031, …) — a chave de setor ali é cod_setor, não cd_setor (confirmado rodando contra os arquivos reais em 25/09/2026; renomeado aqui para cd_setor, o nome usado no resto do pipeline).

# cod_setor precisa ser lido como texto (col_character): sem isso, o readr
# adivinha "número" para um código de 15 dígitos e ele vira double — que
# tanto perde precisão (15 dígitos está no limite do double) quanto quebra
# o join com o cd_setor (texto) do resto do pipeline em silêncio (nenhum
# erro, só 0% de correspondência). Achado rodando a etapa 03 de verdade.
pop01 <- read_csv(
  file.path(CN22_POP_DIR, "cn22_pop01_abs.csv"),
  col_types = cols(cod_setor = col_character(), .default = col_guess())
) |>
  transmute(cd_setor = cod_setor, total = cn22_pop01_res_tot_0)

pop03 <- read_csv(
  file.path(CN22_POP_DIR, "cn22_pop03_abs.csv"),
  col_types = cols(cod_setor = col_character(), .default = col_guess())
) |>
  transmute(
    cd_setor = cod_setor,
    fx_0a4 = cn22_pop03_0a4_tot_0,
    fx_5a9 = cn22_pop03_5a9_tot_0,
    fx_10a14 = cn22_pop03_10a14_tot_0,
    fx_15a19 = cn22_pop03_15a19_tot_0
  )

pop18_setor <- pop01 |>
  left_join(pop03, by = "cd_setor") |>
  mutate(
    across(c(fx_0a4, fx_5a9, fx_10a14, fx_15a19), ~ replace(.x, is.na(.x), 0)),
    pop18 = pmax(total - fx_0a4 - fx_5a9 - fx_10a14 - 0.6 * fx_15a19, 0)
  ) |>
  select(cd_setor, pop18)

write_parquet(pop18_setor, file.path(AUX_DIR, "pop18_setor.parquet"))

4. Setores censitários 2022

O shapefile de setores 2022 do IBGE é grande demais para versionar; espera-se que já esteja em dados/01_bruto/aux/BR_Malha_Preliminar_2022.zip (baixado à mão, ver PLAN.md > Fontes de dados), lido de dentro do zip sem extrair — o mesmo padrão de cnefe-etl/index.qmd (/vsizip/).

caminho_zip <- "dados/01_bruto/aux/BR_Malha_Preliminar_2022.zip"

setores <- st_read(
  paste0("/vsizip/", caminho_zip),
  quiet = TRUE
) |>
  rename(cd_setor = CD_SETOR) |>
  st_transform(4674)

st_write(setores, file.path(AUX_DIR, "setores_2022.gpkg"), delete_dsn = TRUE)
